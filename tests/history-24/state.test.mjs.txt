import {test} from 'node:test';import assert from 'node:assert/strict';import {empty,validate,chooseSituation,toggle,finish} from '../state.js';import {load,save,clear,KEY} from '../storage.js';import {cardText} from '../card.js';
test('две ситуации и выбор источников дают разные содержательные результаты',()=>{let a=chooseSituation(empty(),'work'),b=chooseSituation(empty(),'evening');a=finish(a);b=finish(b);assert.equal(a.done,true);assert.deepEqual(a.lights,['ambient','task']);assert.deepEqual(b.lights,['ambient','accent']);assert.notEqual(cardText(a),cardText(b));assert.match(cardText(a),/электрики/);assert.match(cardText(b),/Тихий вечер/)});
test('непустой выбор, повторное переключение, изменение без ложного финала',()=>{let s=chooseSituation(empty(),'work');s=toggle(toggle(s,'ambient'),'task');assert.equal(finish(s).done,false);s=toggle(s,'accent');s=finish(s);assert.equal(s.done,true);s=toggle(s,'accent');assert.equal(s.done,false);assert.equal(toggle(s,'unknown'),s)});
test('миграция частичных и полных v1/v2/v3 с сохранением старых данных',()=>{const ids=['dawn','focus','arc','pause','line','halo','fold','point'];const correct=[1,2,0,1,2,0,1,2];for(const version of [1,2,3])for(const size of [2,8]){const old={version,answers:Object.fromEntries(ids.slice(0,size).map((id,i)=>[id,version===1?0:correct[i]]))};if(version===3){old.draft={scenario:null,modules:[]};old.personal=null}let s=validate(old);assert.equal(s.version,4);assert.deepEqual(s.legacy,old);assert.equal(s.done,false);assert.deepEqual(validate(s),s)}const full={version:3,answers:Object.fromEntries(ids.map((x,i)=>[x,correct[i]])),draft:{scenario:'rest',modules:['dawn','halo']},personal:{scenario:'rest',modules:['dawn','halo']}};const s=validate(full);assert.equal(s.situation,'evening');assert.deepEqual(s.lights,['ambient','accent']);assert.deepEqual(s.legacy,full)});
test('повреждения и недопустимые значения отклоняются',()=>{for(const x of [null,{version:2,answers:{dawn:0}},{...empty(),lights:['fake']},{...empty(),situation:'work',lights:['task','task']},{...empty(),done:true},{...empty(),legacy:{version:1,answers:{fake:0}}}])assert.equal(validate(x),null)});
test('хранилище и отказ',()=>{const map=new Map();const storage={getItem:k=>map.get(k)??null,setItem:(k,v)=>map.set(k,v),removeItem:k=>map.delete(k)};const s=finish(chooseSituation(empty(),'work'));assert.ok(save(storage,s));assert.deepEqual(load(storage).state,s);map.set(KEY,'{');assert.equal(load(storage).status,'invalid');assert.ok(clear(storage));assert.equal(load(storage).status,'empty');const fail={getItem(){throw 0},setItem(){throw 0},removeItem(){throw 0}};assert.equal(load(fail).status,'unavailable');assert.equal(save(fail,s),false);assert.equal(clear(fail),false)});
